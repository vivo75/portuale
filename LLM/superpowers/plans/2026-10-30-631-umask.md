- [ ] **Step 1: Write the failing test**

```rust
#[test]
fn demoted_spawn_applies_the_portage_umask() {
    if unsafe { libc::geteuid() } != 0 { eprintln!("SKIP: needs root"); return; }
    let Some(ids) = portage_phase_ids(None) else { eprintln!("SKIP: no portage user"); return; };
    // Hostile caller mask with save/restore (UmaskGuard precedent):
    // proves pre_exec sets 022 rather than inheriting it, and proves
    // the parent mask is unchanged afterwards.
    let before = unsafe { libc::umask(0o002) };
    struct Restore(libc::mode_t);
    impl Drop for Restore { fn drop(&mut self) { unsafe { libc::umask(self.0) }; } }
    let _restore = Restore(before);
    let tmp = TempDir::new("portuale-test-droppriv-umask");
    // script prints `umask`, creates a file and a dir, stats both:
    //   umask; touch f; mkdir d; stat -c '%a %n' f d
    // ...through spawn_bash_script with droppriv: Some(ids) (argv_word
    // "compile", Isolation::default(), PhaseLog::Capture) and a second
    // control spawn with droppriv: None.
    // Assert demoted: `umask` line is "0022", f is 644, d is 755.
    // Assert control: `umask` line is "0002" (caller mask untouched).
    // Assert libc::umask(current) reads back `before` (parent unleaked).
}
```

(Reading the parent mask back: `let cur = unsafe { libc::umask(before) };` inside the guard's lifetime is a no-op write of the already-restored value — assert `cur == before`. Keep the whole hostile window to the two spawns; other tests' explicit `chmod` pins are umask-independent.)
