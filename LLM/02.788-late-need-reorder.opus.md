```rust
/// The plan with each late build need moved out of the way when that is
/// legal: the need's target, with the rows between that it reaches over
/// build edges, goes just before the row that needs it, and the move is kept
/// only when it lowers the number of late needs. A plan with a bootstrap row
/// is returned as it is.
///
/// # Errors
/// The repositories or the configuration cannot be read.
pub fn avoid_late_needs<'a>(
    world: &'a World,
    mut plan: BuildPlan<'a>,
    root: &Path,
    config_root: &Path,
) -> Result<(BuildPlan<'a>, Vec<LateNeed>), String> {
    let mut needs = late_build_needs(world, &plan, root, config_root)?;
    if plan.rows.iter().any(|r| r.bootstrap.is_some()) {
        return Ok((plan, needs));
    }
    let mut tried: HashSet<(String, String)> = HashSet::new();
    loop {
        let key = |n: &LateNeed| (n.row.clone(), n.target.clone());
        let Some(k) = needs.iter().position(|n| !tried.contains(&key(n))) else {
            break;
        };
        tried.insert(key(&needs[k]));
        let Some(moved) = moved_before(world, &plan, &needs[k]) else { continue };
        let found = late_build_needs(world, &moved, root, config_root)?;
        if found.len() < needs.len() {
            plan = moved;
            needs = found;
            tried.clear();
        }
    }
    Ok((plan, needs))
}

/// `plan` with the first row of `need.target` after `need.row`, and the rows
/// between them that the target reaches over build edges, moved in their
/// order to just before `need.row`.
fn moved_before<'a>(world: &World, plan: &BuildPlan<'a>, need: &LateNeed) -> Option<BuildPlan<'a>> {
    let i = plan.rows.iter().position(|r| r.slot.node() == need.row)?;
    let j = (i + 1..plan.rows.len()).find(|&j| plan.rows[j].slot.node() == need.target)?;
    let in_block = |k: usize| {
        k == j
            || build_path(&world.edges, &world.edge_kinds, &need.target, &plan.rows[k].slot.node())
                .is_some()
    };
    let row = |k: usize| PlanRow { slot: plan.rows[k].slot, bootstrap: None };
    let block: Vec<usize> = (i + 1..=j).filter(|&k| in_block(k)).collect();
    let mut rows: Vec<PlanRow<'a>> = (0..i).map(row).collect();
    rows.extend(block.iter().map(|&k| row(k)));
    rows.extend((i..plan.rows.len()).filter(|k| !block.contains(k)).map(row));
    Some(BuildPlan { rows, bootstrap: plan.bootstrap })
}
```
